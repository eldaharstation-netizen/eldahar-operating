// ============================================================ // AreaAnalytics_Enhanced.gs — تحليل استهلاك المناطق (نسخة محسّنة) // مع دعم مصدرين منفصلين: العدادات والتصرفات // ============================================================ /** * 🆕 getAreaAnalyticsFromMeters — تحديد استهلاك المنطقة من العدادات فقط * (قراءة البداية والنهاية من شيت "عدادات_المناطق") * * الفوائد: * - مستقل عن بيانات التصرفات الساعية * - يمكن عمل تقرير منفصل للعدادات * - قراءة دقيقة من الأجهزة الفيزيائية */ function getAreaAnalyticsFromMeters(area, from, to) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var fromD = normalizeDate(from), toD = normalizeDate(to); // 🛠️ FIX: لو المنطقة فاضية أو "الكل"، متبقاش بتدور على منطقة اسمها // حرفيًا "الكل" (مش موجودة في الشيت)، لازم تجمع كل المناطق مع بعض var matchAll = !area || area === 'الكل'; var meterSheet = ss.getSheetByName('عدادات_المناطق'); if (!meterSheet || meterSheet.getLastRow() < 2) { return { area: matchAll ? 'كل المناطق' : area, from: fromD, to: toD, source: 'meters', totalConsumption: 0, daysCount: 0, dailyData: [], errorMsg: '❌ شيت العدادات غير موجود أو فارغ' }; } var mrows = meterSheet.getDataRange().getValues(); // الأعمدة: التاريخ[0], اليوم[1], المهندس[2], الخط[3], المنطقة[4], قبل[5], بعد[6], الفرق[7], الوقت[8] var consumptionByDate = {}; var dailyRecords = {}; // للتخزين المؤقت لجميع السجلات لكل يوم for (var k = 1; k < mrows.length; k++) { var mr = mrows[k]; var d = normalizeDate(mr[0]); if (d < fromD || d > toD) continue; var meterArea = String(mr[4] || '').trim(); if (!matchAll && meterArea !== area) continue; var before = parseFloat(mr[5]) || 0; var after = parseFloat(mr[6]) || 0; var diff = parseFloat(mr[7]) || 0; // إذا لم يتم حساب الفرق، احسبه يدويًا if (diff === 0 && (before || after)) { diff = after - before; } consumptionByDate[d] = (consumptionByDate[d] || 0) + diff; if (!dailyRecords[d]) dailyRecords[d] = []; dailyRecords[d].push({ line: String(mr[3] || ''), area: meterArea, before: before, after: after, diff: diff, engineer: String(mr[2] || '') }); } var daily = Object.keys(consumptionByDate).sort().map(function(d) { return { date: d, consumption: Math.round(consumptionByDate[d] * 100) / 100, records: dailyRecords[d].length, details: dailyRecords[d] }; }); var totalConsumption = daily.reduce(function(s, d) { return s + d.consumption; }, 0); return { area: matchAll ? 'كل المناطق' : area, from: fromD, to: toD, source: 'meters', totalConsumption: Math.round(totalConsumption * 100) / 100, daysCount: daily.length, avgDailyConsumption: daily.length ? Math.round(totalConsumption / daily.length * 100) / 100 : 0, daily: daily }; } /** * 🆕 getAreaAnalyticsFromFlows — تحديد استهلاك المنطقة من التصرفات الساعية فقط * (من شيت "البيانات_الساعية") * * الفوائد: * - مستقل عن قراءات العدادات اليدوية * - يمكن عمل تقرير منفصل للتصرفات * - تحليل التصرفات بالساعة */ function getAreaAnalyticsFromFlows(area, from, to) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var fromD = normalizeDate(from), toD = normalizeDate(to); // 🛠️ FIX: نفس إصلاح "الكل" الموجود في getAreaAnalyticsFromMeters var matchAll = !area || area === 'الكل'; var hourSheet = ss.getSheetByName('البيانات_الساعية'); if (!hourSheet || hourSheet.getLastRow() < 2) { return { area: matchAll ? 'كل المناطق' : area, from: fromD, to: toD, source: 'flows', totalConsumption: 0, totalOperatingHours: 0, daysCount: 0, dailyData: [], errorMsg: '❌ شيت البيانات الساعية غير موجود أو فارغ' }; } var hrows = hourSheet.getDataRange().getValues(); var lineDefs = [ { areaIdx: 7, pumpIdx: 8, statusIdx: 9, flowIdx: 12, label: 'خط 1' }, { areaIdx: 14, pumpIdx: 15, statusIdx: 16, flowIdx: 19, label: 'خط 2' }, { areaIdx: 21, pumpIdx: 22, statusIdx: 23, flowIdx: 26, label: 'خط 3' } ]; var pumpHours = {}; var lineHours = {}; var operatingHoursByDate = {}; var consumptionByDate = {}; var totalOperatingHours = 0; for (var k = 1; k < hrows.length; k++) { var hr = hrows[k]; var d2 = normalizeDate(hr[0]); if (d2 < fromD || d2 > toD) continue; lineDefs.forEach(function(ld) { var stopped = hr[ld.statusIdx] === 'stopped' || hr[ld.statusIdx] === 'متوقف'; if (stopped) return; var areasAll = String(hr[ld.areaIdx] || '').split('+').map(function(s) { return s.trim(); }).filter(Boolean); if (!areasAll.length) return; if (!matchAll && areasAll.indexOf(area) === -1) return; totalOperatingHours++; operatingHoursByDate[d2] = (operatingHoursByDate[d2] || 0) + 1; lineHours[ld.label] = (lineHours[ld.label] || 0) + 1; var flow = parseFloat(hr[ld.flowIdx]) || 0; // لو "كل المناطق" خد التصرف الكامل للخط في الساعة دي (من غير تقسيم) // عشان الإجمالي يبقى تصرف المحطة الحقيقي مش مقسوم بلا داعي var share = matchAll ? flow : (flow / areasAll.length); consumptionByDate[d2] = (consumptionByDate[d2] || 0) + share; var pumps = String(hr[ld.pumpIdx] || '').split('+').map(function(s) { return s.trim(); }).filter(Boolean); if (!pumps.length) pumps = ['غير محدد']; pumps.forEach(function(p) { pumpHours[p] = (pumpHours[p] || 0) + 1; }); }); } var dailyDates = {}; Object.keys(operatingHoursByDate).forEach(function(d) { dailyDates[d] = true; }); Object.keys(consumptionByDate).forEach(function(d) { dailyDates[d] = true; }); var daily = Object.keys(dailyDates).sort().map(function(d) { var cons = consumptionByDate[d] || 0; var hrs = operatingHoursByDate[d] || 0; return { date: d, consumption: Math.round(cons * 100) / 100, operatingHours: hrs, avgFlowM3PerHour: hrs ? Math.round(cons / hrs * 100) / 100 : null }; }); var totalConsumption = daily.reduce(function(s, d) { return s + d.consumption; }, 0); return { area: matchAll ? 'كل المناطق' : area, from: fromD, to: toD, source: 'flows', totalConsumption: Math.round(totalConsumption * 100) / 100, totalOperatingHours: totalOperatingHours, daysCount: daily.length, avgDailyConsumption: daily.length ? Math.round(totalConsumption / daily.length * 100) / 100 : 0, avgDailyOperatingHours: daily.length ? Math.round(totalOperatingHours / daily.length * 10) / 10 : 0, avgFlowM3PerHour: totalOperatingHours ? Math.round(totalConsumption / totalOperatingHours * 100) / 100 : 0, pumpHours: pumpHours, lineHours: lineHours, daily: daily }; } /** * 🆕 compareAreaAnalytics — مقارنة الاستهلاك من المصدرين * (العدادات vs التصرفات) لاكتشاف الفروقات */ function compareAreaAnalytics(area, from, to) { var metersData = getAreaAnalyticsFromMeters(area, from, to); var flowsData = getAreaAnalyticsFromFlows(area, from, to); var comparison = { area: area, from: from, to: to, metersTotal: metersData.totalConsumption, flowsTotal: flowsData.totalConsumption, difference: Math.round((metersData.totalConsumption - flowsData.totalConsumption) * 100) / 100, percentDiff: flowsData.totalConsumption > 0 ? Math.round((Math.abs(metersData.totalConsumption - flowsData.totalConsumption) / flowsData.totalConsumption) * 100) : 0, metersData: metersData, flowsData: flowsData }; return comparison; } /** * 🆕 getAllAreasAnalyticsFromMeters — جميع المناطق من العدادات */ function getAllAreasAnalyticsFromMeters(from, to) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var fromD = normalizeDate(from), toD = normalizeDate(to); var meterSheet = ss.getSheetByName('عدادات_المناطق'); var areaStats = {}; if (meterSheet && meterSheet.getLastRow() >= 2) { var mrows = meterSheet.getDataRange().getValues(); for (var k = 1; k < mrows.length; k++) { var mr = mrows[k]; var d = normalizeDate(mr[0]); if (d < fromD || d > toD) continue; var area = String(mr[4] || '').trim(); if (!area) continue; var diff = parseFloat(mr[7]) || 0; if (!areaStats[area]) { areaStats[area] = { consumption: 0, recordsCount: 0 }; } areaStats[area].consumption += diff; areaStats[area].recordsCount++; } } var areas = Object.keys(areaStats).map(function(a) { var s = areaStats[a]; return { area: a, consumption: Math.round(s.consumption * 100) / 100, recordsCount: s.recordsCount, avgPerRecord: s.recordsCount ? Math.round(s.consumption / s.recordsCount * 100) / 100 : 0 }; }).sort(function(a, b) { return b.consumption - a.consumption; }); return { from: fromD, to: toD, source: 'meters', areas: areas }; } /** * 🆕 getAllAreasAnalyticsFromFlows — جميع المناطق من التصرفات */ function getAllAreasAnalyticsFromFlows(from, to) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var fromD = normalizeDate(from), toD = normalizeDate(to); var hourSheet = ss.getSheetByName('البيانات_الساعية'); var areaStats = {}; if (hourSheet && hourSheet.getLastRow() >= 2) { var hrows = hourSheet.getDataRange().getValues(); var lineDefs = [ { areaIdx: 7, statusIdx: 9, flowIdx: 12 }, { areaIdx: 14, statusIdx: 16, flowIdx: 19 }, { areaIdx: 21, statusIdx: 23, flowIdx: 26 } ]; for (var k = 1; k < hrows.length; k++) { var hr = hrows[k]; var d2 = normalizeDate(hr[0]); if (d2 < fromD || d2 > toD) continue; lineDefs.forEach(function(ld) { var stopped = hr[ld.statusIdx] === 'stopped' || hr[ld.statusIdx] === 'متوقف'; if (stopped) return; var areasAll = String(hr[ld.areaIdx] || '').split('+').map(function(s) { return s.trim(); }).filter(Boolean); if (!areasAll.length) return; var flow = parseFloat(hr[ld.flowIdx]) || 0; var share = flow / areasAll.length; areasAll.forEach(function(a) { if (!areaStats[a]) areaStats[a] = { consumption: 0, operatingHours: 0 }; areaStats[a].consumption += share; areaStats[a].operatingHours += 1; }); }); } } var areas = Object.keys(areaStats).map(function(a) { var s = areaStats[a]; return { area: a, consumption: Math.round(s.consumption * 100) / 100, operatingHours: s.operatingHours, avgFlowM3PerHour: s.operatingHours ? Math.round(s.consumption / s.operatingHours * 100) / 100 : null }; }).sort(function(a, b) { return b.consumption - a.consumption; }); return { from: fromD, to: toD, source: 'flows', areas: areas }; } // ============================================================ // دوال مساعدة // ============================================================ function normalizeDate(v) { if (Object.prototype.toString.call(v) === '[object Date]') { return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } var s = String(v).trim(); var mdy = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/); if (mdy) { return mdy[3] + '-' + mdy[1].padStart(2, '0') + '-' + mdy[2].padStart(2, '0'); } return s; } /** * 🆕 createDualAnalyticsSheet — إنشاء شيت تقرير يعرض الاستهلاك من المصدرين معاً */ function createDualAnalyticsSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetName = 'تحليل_المناطق_مقارن'; var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } else { sheet.clear(); } sheet.appendRow([ 'المنطقة', 'الاستهلاك من العدادات (م³)', 'الاستهلاك من التصرفات (م³)', 'الفرق (م³)', 'نسبة الفرق (%)', 'ملاحظات' ]); sheet.getRange(1, 1, 1, 6) .setBackground('#1a6b3c') .setFontColor('white') .setFontWeight('bold'); sheet.setFrozenRows(1); return sheet; } /** * 🆕 populateDualAnalytics — ملء تقرير المقارنة بيانات لفترة معينة */ function populateDualAnalytics(from, to, sheetName) { sheetName = sheetName || 'تحليل_المناطق_مقارن'; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = createDualAnalyticsSheet(); } if (sheet.getLastRow() > 1) { sheet.deleteRows(2, sheet.getLastRow() - 1); } var metersAnalytics = getAllAreasAnalyticsFromMeters(from, to); var flowsAnalytics = getAllAreasAnalyticsFromFlows(from, to); var metersByArea = {}; metersAnalytics.areas.forEach(function(a) { metersByArea[a.area] = a.consumption; }); var flowsByArea = {}; flowsAnalytics.areas.forEach(function(a) { flowsByArea[a.area] = a.consumption; }); var allAreas = Object.keys({ ...metersByArea, ...flowsByArea }); var rows = []; allAreas.forEach(function(area) { var meterVal = metersByArea[area] || 0; var flowVal = flowsByArea[area] || 0; var diff = Math.round((meterVal - flowVal) * 100) / 100; var percentDiff = flowVal > 0 ? Math.round((Math.abs(diff) / flowVal) * 100) : 0; var note = ''; if (Math.abs(diff) > 100) { note = '⚠ فرق كبير — راجع البيانات'; } else if (Math.abs(diff) > 50) { note = '⚡ فرق متوسط'; } rows.push([area, meterVal, flowVal, diff, percentDiff + '%', note]); }); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, 6).setValues(rows); } sheet.autoResizeColumns(1, 6); return sheet; }
